Sinkronisasi Google Sheets (Wrapped2026)

Mode Demo Lokal

Setiap pelari yang menekan tombol unduh otomatis masuk ke tab sheet Wrapped2026 dan langsung tampil di Rekap Komunitas.

Code.gs — Sheet "Wrapped2026" (doPost & doGet) 
function getOrCreateSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Wrapped2026");
  if (!sheet) {
    sheet = ss.insertSheet("Wrapped2026");
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Timestamp", "Nama", "Instagram", "Cabang",
      "TotalMedali", "TotalKM", "Gelar",
      "DaftarRace", "Target2027", "SiapReview"
    ]);
  }
  return sheet;
}

function doPost(e) {
  try {
    var sheet = getOrCreateSheet_();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date().toISOString(),
      data.name || "",
      data.instagram || "",
      data.cabang || "Kansai",
      Number(data.medalCount || 0),
      Number(data.totalKm || 0),
      data.persona || "",
      JSON.stringify(data.races || []),
      data.target2027 || "",
      data.openToReview ? "Ya" : "Tidak"
    ]);
    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = getOrCreateSheet_();
    var values = sheet.getDataRange().getValues();
    var entries = [];
    for (var i = 1; i < values.length; i++) {
      var row = values[i];
      if (!row[1]) continue;
      var parsedRaces = [];
      try {
        parsedRaces = JSON.parse(row[7] || "[]");
      } catch (err) {
        parsedRaces = String(row[7] || "").split(";").map(function(s) { return s.trim(); }).filter(Boolean);
      }
      entries.push({
        id: "sheet-" + i,
        timestamp: row[0] ? String(row[0]) : "",
        name: String(row[1] || ""),
        instagram: String(row[2] || ""),
        cabang: String(row[3] || "Kansai"),
        medalCount: Number(row[4] || parsedRaces.length || 0),
        totalKm: Number(row[5] || 0),
        persona: String(row[6] || "All-Rounder 2026"),
        races: Array.isArray(parsedRaces) ? parsedRaces : [],
        target2027: String(row[8] || ""),
        openToReview: String(row[9]).toLowerCase() === "ya" || row[9] === true
      });
    }
    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok", entries: entries.reverse() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
